Mahadev Fashion Hub - Local Backend Server
Handles all GitHub operations automatically. Admin panel never sees the token.

import os
import json
import base64
import subprocess
from http.server import HTTPServer, SimpleHTTPRequestHandler
from urllib.request import Request, urlopen
from urllib.error import HTTPError
import cloudinary
import cloudinary.uploader
from multipart import parse_form_data

ENV_FILE = ".env"
PRODUCTS = "products.json"
PORT = 5000


# ─────────────────────────────────────────────
# Load .env
# ─────────────────────────────────────────────
def load_env():
    env = {}
    if os.path.exists(ENV_FILE):
        with open(ENV_FILE, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if line and not line.startswith("#") and "=" in line:
                    k, v = line.split("=", 1)
                    env[k.strip()] = v.strip().strip('"').strip("'")
    return env


ENV = load_env()
GH_USER = ENV.get("GITHUB_USER", "")
GH_REPO = ENV.get("GITHUB_REPO", "")
GH_TOKEN = ENV.get("GITHUB_TOKEN", "")
GH_BRANCH = ENV.get("GITHUB_BRANCH", "main")
cloudinary.config(
    cloud_name=ENV.get("CLOUDINARY_CLOUD_NAME"),
    api_key=ENV.get("CLOUDINARY_API_KEY"),

    secure=True
)

def gh_get_sha():
    """Get current products.json SHA from GitHub (needed for update)."""
    url = f"https://api.github.com/repos/{GH_USER}/{GH_REPO}/contents/{PRODUCTS}?ref={GH_BRANCH}"
    req = Request(url, headers={
        "Authorization": f"Bearer {GH_TOKEN}",
        "Accept": "application/vnd.github+json"
    })
    try:
        with urlopen(req) as r:
            return json.loads(r.read())["sha"]
    except HTTPError as e:
        if e.code == 404:
            return None
        raise


def gh_push(content_str, commit_msg):
    """Push products.json to GitHub via API."""
    sha = gh_get_sha()
    content_b64 = base64.b64encode(content_str.encode("utf-8")).decode("ascii")

    body = {
        "message": commit_msg,
        "content": content_b64,
        "branch": GH_BRANCH
    }
    if sha:
        body["sha"] = sha

    url = f"https://api.github.com/repos/{GH_USER}/{GH_REPO}/contents/{PRODUCTS}"
    req = Request(url, method="PUT", data=json.dumps(body).encode("utf-8"), headers={
        "Authorization": f"Bearer {GH_TOKEN}",
        "Accept": "application/vnd.github+json",
        "Content-Type": "application/json"
    })
    with urlopen(req) as r:
        return json.loads(r.read())

class Handler(SimpleHTTPRequestHandler):
    def _cors(self):
        self.send_header("Access-Control-Allow-Origin", "*")
        self.send_header("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
        self.send_header("Access-Control-Allow-Headers", "Content-Type")

    def do_OPTIONS(self):
        self.send_response(200)
        self._cors()
        self.end_headers()

    def do_GET(self):
        # Health check
        if self.path == "/health":
            self.send_response(200)
            self.send_header("Content-Type", "application/json")
            self._cors()
            self.end_headers()
            self.wfile.write(json.dumps({
                "ok": True,
                "user": GH_USER,
                "repo": GH_REPO,
                "branch": GH_BRANCH,
                "token_set": bool(GH_TOKEN)
            }).encode())
            return
        # Serve static files normally (admin.html, index.html, etc.)
        return super().do_GET()

    def do_POST(self):
        # ── Upload image to Cloudinary ──
        if self.path == "/upload":
            try:
                content_type = self.headers.get("Content-Type", "")
                content_length = int(self.headers.get("Content-Length", 0))
                
                if "boundary=" not in content_type:
                    self._json(400, {"ok": False, "error": "Missing boundary"})
                    return
                
                boundary = content_type.split("boundary=")[1].strip()
                if boundary.startswith('"') and boundary.endswith('"'):
                    boundary = boundary[1:-1]
                
                from multipart import MultipartParser
                import io
                
                body = self.rfile.read(content_length)
                parser = MultipartParser(io.BytesIO(body), boundary, charset="latin-1")
                
                # Collect ALL files from the multipart body
                uploaded = []
                for part in parser:
                    if part.name == "file":
                        raw = part.value if isinstance(part.value, bytes) else part.value.encode('latin-1')
                        fname = (part.filename or "upload.jpg").replace('\x00', '').strip()
                        if isinstance(fname, bytes):
                            fname = fname.decode('latin-1', errors='ignore')
                        if not fname:
                            fname = "upload.jpg"
                        if '.' not in fname:
                            fname = fname + '.jpg'
                        uploaded.append((raw, fname))
                
                if not uploaded:
                    self._json(400, {"ok": False, "error": "No files received"})
                    return
                
                if len(uploaded) > 5:
                    self._json(400, {"ok": False, "error": "Max 5 images per product"})
                    return
                
                # Read optional category + product_id from form fields
                # (sent by admin.html alongside the file inputs)
                category = ""
                product_id = ""
                for part in parser:
                    if part.name == "category":
                        category = (part.value or "").strip() if not isinstance(part.value, bytes) else part.value.decode("latin-1").strip()
                    elif part.name == "product_id":
                        product_id = (part.value or "").strip() if not isinstance(part.value, bytes) else part.value.decode("latin-1").strip()
                
                # Sanitize for folder use
                category = "".join(c for c in category if c.isalnum() or c in "-_")
                product_id = "".join(c for c in product_id if c.isalnum() or c in "-_")
                
                if not category:
                    category = "Uncategorized"
                if not product_id:
                    product_id = "misc"
                
                folder_path = f"ma-fashion/{category}/{product_id}"
                
                urls = []
                for raw, fname in uploaded:
                    result = cloudinary.uploader.upload(
                        raw,
                        asset_folder=folder_path,
                        use_asset_folder_as_public_id_prefix=True,
                        overwrite=False,
                        resource_type="image"
                    )
                    urls.append(result["secure_url"])
                
                self._json(200, {"ok": True, "urls": urls, "url": urls[0]})
                
            except Exception as e:
                self._json(500, {"ok": False, "error": str(e)})
            return
                
        # ── Save catalog → local + GitHub (auto) ──
        if self.path == "/save":
            length = int(self.headers.get("Content-Length", 0))
            raw = self.rfile.read(length).decode("utf-8")

            try:
                data = json.loads(raw)
            except Exception:
                self._json(400, {"ok": False, "error": "Invalid JSON"})
                return

            # 1. Write locally
            with open(PRODUCTS, "w", encoding="utf-8") as f:
                json.dump(data, f, indent=2, ensure_ascii=False)

            # 2. Push to GitHub
            try:
                pretty = json.dumps(data, indent=2, ensure_ascii=False)
                commit_msg = f"Admin update: {len(data)} products"
                result = gh_push(pretty, commit_msg)
                self._json(200, {
                    "ok": True,
                    "message": "Saved & pushed to GitHub",
                    "commit": result.get("commit", {}).get("sha", "")[:7]
                })
            except HTTPError as e:
                err_body = e.read().decode()
                self._json(500, {"ok": False, "error": f"GitHub: {err_body}"})
            except Exception as e:
                self._json(500, {"ok": False, "error": str(e)})
            return

        self._json(404, {"ok": False, "error": "Not found"})

    def _json(self, code, obj):
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self._cors()
        self.end_headers()
        self.wfile.write(json.dumps(obj).encode())

    def log_message(self, fmt, *args):
        print(f"[server] {self.address_string()} - {fmt % args}")

import os

PATCH_FILE = "patch.txt"
ENV_FILE   = ".env"
PRODUCTS   = "products.json"
QUEUE_FILE = "pending_catalog.json"


# ─────────────────────────────────────────────────────
# Load .env (no external deps)
# ─────────────────────────────────────────────────────
def load_env():
    if not os.path.exists(ENV_FILE):
        return {}
    env = {}
    with open(ENV_FILE, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line or line.startswith("#") or "=" not in line:
                continue
            k, v = line.split("=", 1)
            env[k.strip()] = v.strip().strip('"').strip("'")
    return env


# ─────────────────────────────────────────────────────
# Patch.txt parser (find / replace)
# ─────────────────────────────────────────────────────
def parse_patch_file(filepath):
    if not os.path.exists(filepath):
        print(f"❌ '{filepath}' not found.")
        return []
    with open(filepath, "r", encoding="utf-8") as f:
        content = f.read()

    patches = []
    for sec in content.split("FILE:"):
        sec = sec.strip()
        if not sec:
            continue
        lines = sec.splitlines()
        filename = lines[0].strip()
        remaining = "\n".join(lines[1:])
        for blk in remaining.split("<<<<<<< FIND"):
            if "=======" in blk and ">>>>>>>" in blk:
                find_part, rest = blk.split("=======", 1)
                replace_part, _ = rest.split(">>>>>>>", 1)
                find_str = find_part.strip("\r\n")
                replace_str = replace_part.strip("\r\n")
                if find_str:
                    patches.append({"file": filename,
                                    "find": find_str,
                                    "replace": replace_str})
    return patches


def apply_patches(patches):
    if not patches:
        print("⚠️  No patches to apply.")
        return 0
    count = 0
    for p in patches:
        tf = p["file"]
        if not os.path.exists(tf):
            print(f"❌ Missing file: {tf}")
            continue
        with open(tf, "r", encoding="utf-8") as f:
            data = f.read().replace("\r\n", "\n")
        find    = p["find"].replace("\r\n", "\n")
        replace = p["replace"].replace("\r\n", "\n")
        if find in data:
            with open(tf, "w", encoding="utf-8", newline="\n") as f:
                f.write(data.replace(find, replace, 1))
            print(f"✅ Patched [{tf}]")
            count += 1
        else:
            print(f"❌ No match in [{tf}] → {find[:60]!r}")
    return count


# ─────────────────────────────────────────────────────
# Apply admin queue → products.json
# ─────────────────────────────────────────────────────
def apply_admin_queue():
    if not os.path.exists(QUEUE_FILE):
        return False
    try:
        with open(QUEUE_FILE, "r", encoding="utf-8") as f:
            incoming = json.load(f)
    except Exception as e:
        print(f"⚠️  Bad queue file: {e}")
        return False

    with open(PRODUCTS, "w", encoding="utf-8") as f:
        json.dump(incoming, f, indent=2, ensure_ascii=False)
    os.remove(QUEUE_FILE)
    print(f"✅ Applied {len(incoming)} product(s) from admin queue → {PRODUCTS}")
    return True

def git_push(commit_msg="Update store via auto-patch"):
    print("\n🚀 Pushing to GitHub...")
    try:
        subprocess.run(["git", "add", "."], check=True)
        res = subprocess.run(["git", "status", "--porcelain"],
                             capture_output=True, text=True)
        if not res.stdout.strip():
            print("ℹ️  Nothing to commit.")
            return
        subprocess.run(["git", "commit", "-m", commit_msg], check=True)
        subprocess.run(["git", "push", "origin", "main"], check=True)
        print("🎉 Pushed! GitHub Pages live in ~20s.")
    except subprocess.CalledProcessError as e:
        print("❌ Git error:", e)

def menu():
    print("=" * 55)
    print("🛠️   MAHADEV FASHION HUB — BACKEND SYNC")
    print("=" * 55)
    print("  1.  Update files  +  Push to Git")
    print("  2.  Update files only  (no push)")

def main():
    env = load_env()
    if not env.get("GITHUB_USER") or not env.get("GITHUB_REPO"):
        print("⚠️  .env missing GITHUB_USER / GITHUB_REPO.")
        print("    Create .env next to this script (see instructions).")

    choice = menu()
    if choice not in ("1", "2", "3"):
        print("❌ Invalid.")
        return

    # 1 & 2 → apply patches + admin queue
    if choice in ("1", "2"):
        patches = parse_patch_file(PATCH_FILE)
        if patches:
            print(f"\n📦 {len(patches)} patch block(s)...")
            apply_patches(patches)
            ans = input("🧹 Clear patch.txt? (Y/n): ").strip().lower()
            if ans != "n":
                open(PATCH_FILE, "w").close()
                print("   Cleared.")
        else:
            print("ℹ️  patch.txt empty.")

        if apply_admin_queue():
            print("📥 Admin queue applied.")

    # 1 & 3 → git push
    if choice in ("1", "3"):
        msg = input("\n📝 Commit message (Enter = default): ").strip() \
              or "Update store via backend sync"
        git_push(msg)

    print("\n🏁 Done. Press Enter to close.")
    input()


if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print("\n⚠️  Cancelled.")

    async function saveProduct() {
      const title = document.getElementById("pTitle").value.trim();
      const price = Number(document.getElementById("pPrice").value);
      const mrp   = Number(document.getElementById("pMrp").value) || price;
      const category = document.getElementById("pCat").value;
      const freeTag  = document.getElementById("pTag").value.trim();
      const desc  = document.getElementById("pDesc").value.trim();
      const editId = document.getElementById("editProductId").value;
      const saveBtn = document.getElementById("saveBtn");
      const original = saveBtn.innerText;

      if (!title || !price) {
        alert("Please enter product name and selling price.");
        return;
      }

      const selectedTags = getSelectedTags();

      // 1. Upload new images (if any were picked)
      let uploadedUrls = [];
      if (currentImageFiles.length > 0) {
        saveBtn.innerText = "⏳ Uploading " + currentImageFiles.length + " image(s)...";
        saveBtn.disabled = true;
        try {
          const form = new FormData();
          currentImageFiles.forEach(f => form.append("file", f));
          form.append("category", category);
          form.append("product_id", editId || ("AJ-" + Date.now().toString().slice(-4)));
          const upRes = await fetch("http://localhost:5000/upload", {
            method: "POST",
            body: form
          });
          const upData = await upRes.json();
          if (!upData.ok) throw new Error(upData.error);
          uploadedUrls = upData.urls || [upData.url];
        } catch (e) {
          alert("❌ Image upload failed: " + e.message);
          saveBtn.innerText = original;
          saveBtn.disabled = false;
          return;
        }
      }

      // Combine: newly uploaded + kept existing (edit mode)
      const finalImages = [...uploadedUrls, ...existingImageUrls];

      if (finalImages.length === 0) {
        alert("Please select at least one image.");
        saveBtn.innerText = original;
        saveBtn.disabled = false;
        return;
      }

      // 2. Build product object
      const productObj = {
        id: editId || "AJ-" + Date.now().toString().slice(-4),
        title,
        price,
        originalPrice: mrp,
        category,
        tag: freeTag,
        tags: selectedTags,
        description: desc,
        image: finalImages[0],       // backwards compat (first image)
        images: finalImages          // new multi-image array
      };

      if (editId) {
        const idx = productsList.findIndex(p => p.id === editId);
        if (idx !== -1) productsList[idx] = productObj;
      } else {
        productsList.unshift(productObj);
      }

      renderCatalog();
      resetForm();
      await pushChangesToGitHub("Update product: " + title);
    }
    /* [END SAVE] */

/* ═══════════════════════════════════════════════════════════════ */
/* [FETCH] Load products.json                                      */
/* ═══════════════════════════════════════════════════════════════ */
async function fetchProducts() {
  try {
    const res = await fetch(`products.json?_t=${Date.now()}`);
    products = await res.json();

    // Backwards compatibility: if a product has old single "tag"
    // but no "tags" array, convert it.
    products.forEach(p => {
      if (!Array.isArray(p.tags)) {
        p.tags = p.tag ? [String(p.tag).toLowerCase().replace(/\s+/g, "_")] : [];
      }
    });

    renderFiltered();
    initCarousel();
  } catch (e) {
    console.error("Could not load products.json", e);
  }
}
/* [END FETCH] */


/* ═══════════════════════════════════════════════════════════════ */
/* [SCRAMBLE] Sort products: bestsellers first, rest random        */
/* -------------------------------------------------------------- */
/* Order rules when nothing is filtered:                           */
/*   1. Products tagged "bestseller" (in original order)           */
/*   2. Products tagged "most_loved"                               */
/*   3. Everything else — shuffled randomly                        */
/* ═══════════════════════════════════════════════════════════════ */
function scrambleProducts(list) {
  const bestsellers = list.filter(p => p.tags.includes("bestseller"));
  const mostLoved   = list.filter(p => p.tags.includes("most_loved") && !p.tags.includes("bestseller"));
  const rest        = list.filter(p =>
    !p.tags.includes("bestseller") && !p.tags.includes("most_loved")
  );

  // Fisher–Yates shuffle on "rest"
  for (let i = rest.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [rest[i], rest[j]] = [rest[j], rest[i]];
  }

  return [...bestsellers, ...mostLoved, ...rest];
}
/* [END SCRAMBLE] */


/* ═══════════════════════════════════════════════════════════════ */
/* [FILTER LOGIC] Combine tag chip + category pill                 */
/* ═══════════════════════════════════════════════════════════════ */
function getFilteredProducts() {
  let list = [...products];

  // Category filter
  if (activeCat !== "all") {
    list = list.filter(p => p.category === activeCat);
  }

  // Tag filter
  if (activeTag !== "all") {
    list = list.filter(p => p.tags.includes(activeTag));
  }

  // Sort: only scramble when both filters are "all"
  if (activeTag === "all" && activeCat === "all") {
    list = scrambleProducts(list);
  }

  return list;
}

function renderFiltered() {
  renderProducts(getFilteredProducts());
}
/* [END FILTER LOGIC] */


/* ═══════════════════════════════════════════════════════════════ */
/* [RENDER] Draw product grid                                      */
/* ═══════════════════════════════════════════════════════════════ */
function renderProducts(items) {
  const container = document.getElementById("productsContainer");

  if (items.length === 0) {
    container.innerHTML = `<p style="grid-column:1/-1;color:var(--text-muted);text-align:center;padding:40px 0;">No products match this filter.</p>`;
    return;
  }

  container.innerHTML = items.map(p => {
    // Pick the first tag for the badge (prefer bestseller)
    const badge = p.tags.includes("bestseller") ? "Bestseller"
                : p.tags.includes("most_loved") ? "Most Loved"
                : p.tags.includes("trending")   ? "Trending"
                : p.tags.includes("new")        ? "New"
                : p.tags.includes("sale")       ? "Sale"
                : (p.tag || "");

    return `
    <div class="product-card" id="card-${p.id}">
      <div class="img-box">
        <img src="${p.image}" alt="${p.title}" loading="lazy">
        ${badge ? `<span class="product-badge">${badge}</span>` : ""}
      </div>
      <div class="product-info">
        <div>
          <h4>${p.title}</h4>
          <div class="price-row">
            <span class="current-price">₹${p.price.toLocaleString("en-IN")}</span>
            <span class="mrp-price">₹${p.originalPrice.toLocaleString("en-IN")}</span>
          </div>
        </div>
        <button class="btn-add" onclick="addToCart('${p.id}')">Add to Bag</button>
      </div>
    </div>
  `;
  }).join("");
}
/* [END RENDER] */


/* ═══════════════════════════════════════════════════════════════ */
/* [CAROUSEL] Auto-sliding hero with dots                          */
/* -------------------------------------------------------------- */
/* Slide order: one product from each category, round-robin.       */
/* Example with 3 categories and 3 products each:                  */
/*   CatA#1 → CatB#1 → CatC#1 → CatA#2 → CatB#2 → CatC#2 → ...    */
/* ═══════════════════════════════════════════════════════════════ */
function buildCarouselOrder() {
  // Group products by category
  const byCategory = {};
  products.forEach(p => {
    if (!byCategory[p.category]) byCategory[p.category] = [];
    byCategory[p.category].push(p);
  });

  const categories = Object.keys(byCategory);
  const maxLen = Math.max(...categories.map(c => byCategory[c].length));
  const ordered = [];

  // Round-robin: pick 1st from each category, then 2nd, then 3rd…
  for (let i = 0; i < maxLen; i++) {
    categories.forEach(cat => {
      if (byCategory[cat][i]) ordered.push(byCategory[cat][i]);
    });
  }

  return ordered;
}

function initCarousel() {
  const track = document.getElementById("carouselTrack");
  const dots  = document.getElementById("carouselDots");
  if (!track || !dots) return;

  carouselSlides = buildCarouselOrder();

  if (carouselSlides.length === 0) {
    track.innerHTML = `<div class="carousel-loading">No products yet</div>`;
    return;
  }

  // Build slides
  track.innerHTML = carouselSlides.map(p => `
    <div class="carousel-slide" data-product-id="${p.id}">
      <img src="${p.image}" alt="${p.title}" loading="eager">
      <div class="carousel-slide-caption">
        <h3>${p.title}</h3>
        <div class="slide-price">₹${p.price.toLocaleString("en-IN")}</div>
      </div>
    </div>
  `).join("");

  // Build dots
  dots.innerHTML = carouselSlides.map((_, i) =>
    `<button class="carousel-dot${i === 0 ? " active" : ""}" data-slide="${i}" aria-label="Slide ${i+1}"></button>`
  ).join("");

  carouselIndex = 0;
  updateCarouselPosition();

  // Slide click → scroll to product card
  track.querySelectorAll(".carousel-slide").forEach(slide => {
    slide.addEventListener("click", () => {
      const id = slide.dataset.productId;
      scrollToProduct(id);
    });
  });

  // Dot click → jump to slide
  dots.querySelectorAll(".carousel-dot").forEach(dot => {
    dot.addEventListener("click", (e) => {
      e.stopPropagation();
      carouselIndex = Number(dot.dataset.slide);
      updateCarouselPosition();
      restartCarouselTimer();
    });
  });

  // Pause on hover / touch
  const carousel = document.getElementById("heroCarousel");
  carousel.addEventListener("mouseenter", stopCarouselTimer);
  carousel.addEventListener("mouseleave", startCarouselTimer);
  carousel.addEventListener("touchstart", stopCarouselTimer, { passive: true });
  carousel.addEventListener("touchend",   startCarouselTimer, { passive: true });

  startCarouselTimer();
}

function updateCarouselPosition() {
  const track = document.getElementById("carouselTrack");
  if (!track) return;
  track.style.transform = `translateX(-${carouselIndex * 100}%)`;

  document.querySelectorAll(".carousel-dot").forEach((dot, i) => {
    dot.classList.toggle("active", i === carouselIndex);
  });
}

function startCarouselTimer() {
  stopCarouselTimer();
  if (carouselSlides.length < 2) return;
  carouselTimer = setInterval(() => {
    carouselIndex = (carouselIndex + 1) % carouselSlides.length;
    updateCarouselPosition();
  }, CAROUSEL_INTERVAL);
}

function stopCarouselTimer() {
  if (carouselTimer) {
    clearInterval(carouselTimer);
    carouselTimer = null;
  }
}

function restartCarouselTimer() {
  stopCarouselTimer();
  startCarouselTimer();
}

function scrollToProduct(productId) {
  // If the product isn't visible under the current filter, reset filters
  const visible = getFilteredProducts().some(p => p.id === productId);
  if (!visible) {
    activeTag = "all";
    activeCat = "all";
    document.querySelectorAll(".filter-chip").forEach(c =>
      c.classList.toggle("active", c.dataset.tag === "all")
    );
    document.querySelectorAll(".cat-pill").forEach(c =>
      c.classList.toggle("active", c.dataset.cat === "all")
    );
    renderFiltered();
  }

  const card = document.getElementById(`card-${productId}`);
  if (!card) return;

  card.scrollIntoView({ behavior: "smooth", block: "center" });
  card.classList.remove("highlight");
  void card.offsetWidth;          // force reflow to restart animation
  card.classList.add("highlight");
  setTimeout(() => card.classList.remove("highlight"), 1900);
}
/* [END CAROUSEL] */


/* ═══════════════════════════════════════════════════════════════ */
/* [CART] Add / remove / update                                    */
/* ═══════════════════════════════════════════════════════════════ */
function addToCart(productId) {
  const item = products.find(p => p.id === productId);
  if (!item) return;

  const inCart = cart.find(c => c.id === productId);
  if (inCart) {
    inCart.qty += 1;
  } else {
    cart.push({ ...item, qty: 1 });
  }
  updateCartUI();
}

function updateCartUI() {
  document.getElementById("cartCount").innerText = cart.reduce((sum, i) => sum + i.qty, 0);

  const cartList = document.getElementById("cartItemsList");
  if (cart.length === 0) {
    cartList.innerHTML = `<p class="empty-msg">Your shopping bag is empty.</p>`;
  } else {
    cartList.innerHTML = cart.map((i, index) => `
      <div class="cart-item">
        <div>
          <strong>${i.title}</strong>
          <div>₹${i.price} × ${i.qty}</div>
        </div>
        <button onclick="removeFromCart(${index})" style="background:none; border:none; color:red; cursor:pointer;">✕</button>
      </div>
    `).join("");
  }

  const total = cart.reduce((sum, i) => sum + (i.price * i.qty), 0);
  document.getElementById("cartTotalDisplay").innerText = `₹${total.toLocaleString("en-IN")}`;
}

function removeFromCart(index) {
  cart.splice(index, 1);
  updateCartUI();
}
/* [END CART] */

/* [LISTENERS] Wire up all UI events                               */
/* ═══════════════════════════════════════════════════════════════ */
function setupListeners() {
  // Cart drawers
  document.getElementById("cartOpenBtn").onclick  = () => document.getElementById("cartOverlay").classList.add("open");
  document.getElementById("cartCloseBtn").onclick = () => document.getElementById("cartOverlay").classList.remove("open");
  document.getElementById("upiCloseBtn").onclick  = () => document.getElementById("upiModal").classList.remove("open");

  // Filter chips
  document.querySelectorAll(".filter-chip").forEach(btn => {
    btn.onclick = () => {
      document.querySelectorAll(".filter-chip").forEach(b => b.classList.remove("active"));
      btn.classList.add("active");
      activeTag = btn.dataset.tag;
      renderFiltered();
    };
  });

  // Category pills
  document.querySelectorAll(".cat-pill").forEach(btn => {
    btn.onclick = () => {
      document.querySelectorAll(".cat-pill").forEach(b => b.classList.remove("active"));
      btn.classList.add("active");
      activeCat = btn.dataset.cat;
      renderFiltered();
    };
  });

  // Checkout
  document.getElementById("checkoutBtn").onclick = () => {
    if (cart.length === 0) return alert("Your bag is empty!");
    const name  = document.getElementById("custName").value.trim();
    const phone = document.getElementById("custPhone").value.trim();
    const addr  = document.getElementById("custAddress").value.trim();

    if (!name || !phone || !addr) return alert("Please fill in your name, phone, and address.");

    const total = cart.reduce((sum, i) => sum + (i.price * i.qty), 0);
    const orderId = "AJ-" + Math.floor(1000 + Math.random() * 9000);

    pendingOrder = { orderId, name, phone, addr, total, items: [...cart] };
    openUpiModal(pendingOrder);
  };
}
/* [END LISTENERS] */


/* ═══════════════════════════════════════════════════════════════ */
/* [UPI] Payment modal                                             */
/* ═══════════════════════════════════════════════════════════════ */
function openUpiModal(order) {
  const upiUri = `upi://pay?pa=${encodeURIComponent(CONFIG.upiId)}&pn=${encodeURIComponent(CONFIG.storeName)}&am=${order.total}&cu=INR&tn=${encodeURIComponent("Order " + order.orderId)}`;
  const qrUrl  = `https://api.qrserver.com/v1/create-qr-code/?size=220x220&data=${encodeURIComponent(upiUri)}`;

  document.getElementById("upiQrImg").src            = qrUrl;
  document.getElementById("upiAmountLabel").innerText = `₹${order.total.toLocaleString("en-IN")}`;
  document.getElementById("upiIdLabel").innerText     = CONFIG.upiId;
  document.getElementById("directUpiLink").href       = upiUri;

  document.getElementById("copyUpiBtn").onclick = () => {
    navigator.clipboard.writeText(CONFIG.upiId);
    alert("UPI ID copied!");
  };

  document.getElementById("proceedWhatsAppBtn").onclick = () => triggerWhatsApp(order);

  document.getElementById("cartOverlay").classList.remove("open");
  document.getElementById("upiModal").classList.add("open");
}
/* [END UPI] */


/* ═══════════════════════════════════════════════════════════════ */
/* [WHATSAPP] Order confirmation                                   */
/* ═══════════════════════════════════════════════════════════════ */
function triggerWhatsApp(order) {
  const itemList = order.items.map(i => `• ${i.title} (x${i.qty}) - ₹${i.price * i.qty}`).join("\n");
  const msg = `🛍️ *NEW ORDER: #${order.orderId}*\n` +
              `---------------------------------\n` +
              `${itemList}\n` +
              `---------------------------------\n` +
              `💰 *Total Amount:* ₹${order.total}\n` +
              `💳 *Payment Method:* UPI (${CONFIG.upiId})\n\n` +
              `📍 *Delivery Details:*\n` +
              `Name: ${order.name}\n` +
              `Phone: ${order.phone}\n` +
              `Address: ${order.addr}\n\n` +
              `✅ *I am attaching my payment screenshot below.*`;

  window.open(`https://wa.me/${CONFIG.whatsappNumber}?text=${encodeURIComponent(msg)}`, "_blank");
  document.getElementById("upiModal").classList.remove("open");
  cart = [];
  updateCartUI();
}
/* [END WHATSAPP] */
